$ cat crud.sh   # verbatim step, localhost:3000 -> frontend:8080
set -euo pipefail
api=http://frontend:8080/api
curl -fsS http://frontend:8080/ | grep -qi '<title>CareQueue'
curl -fsS "${api}/info" | tee /tmp/info.json; echo
jq -e --arg sha "${GITHUB_SHA}" '.commit == $sha' /tmp/info.json

# Create a doctor
doc=$(curl -fsS -X POST "${api}/doctors" -H 'Content-Type: application/json' \
  -d '{"full_name":"Dr. CI Pipeline","specialty":"General Medicine","room":"CI-1"}' | jq -r .id)
curl -fsS "${api}/doctors/${doc}" | jq -e '.full_name == "Dr. CI Pipeline"'

# Create a patient (fake, run-specific Indian mobile number)
phone="9$(printf '%09d' $(( GITHUB_RUN_ID % 1000000000 )))"
pat=$(curl -fsS -X POST "${api}/patients" -H 'Content-Type: application/json' \
  -d "{\"full_name\":\"CI Test Patient\",\"phone\":\"${phone}\",\"date_of_birth\":\"1990-01-15\",\"gender\":\"other\"}" | jq -r .id)
curl -fsS -X PUT "${api}/patients/${pat}" -H 'Content-Type: application/json' \
  -d "{\"full_name\":\"CI Test Patient Updated\",\"phone\":\"${phone}\",\"date_of_birth\":\"1990-01-15\",\"gender\":\"other\"}" \
  | jq -e '.full_name == "CI Test Patient Updated"'

# Book tomorrow 10:00 (clinic time), then the same slot again must be rejected (409)
day=$(TZ=Asia/Kolkata date -d tomorrow +%F)
body="{\"patient_id\":${pat},\"doctor_id\":${doc},\"appointment_date\":\"${day}\",\"slot_time\":\"10:00:00\",\"reason\":\"CI check-up\"}"
appt=$(curl -fsS -X POST "${api}/appointments" -H 'Content-Type: application/json' -d "${body}")
echo "${appt}" | jq -e '.status == "booked" and .token_number >= 1'
id=$(echo "${appt}" | jq -r .id)
test "$(curl -s -o /dev/null -w '%{http_code}' -X POST "${api}/appointments" -H 'Content-Type: application/json' -d "${body}")" = 409
curl -fsS "${api}/doctors/${doc}/slots?date=${day}" | jq -e '[.slots[] | select(.status == "booked")] | length == 1'

# Status transition, then delete
curl -fsS -X PATCH "${api}/appointments/${id}/status" -H 'Content-Type: application/json' \
  -d '{"status":"cancelled"}' | jq -e '.status == "cancelled"'
test "$(curl -s -o /dev/null -w '%{http_code}' -X DELETE "${api}/appointments/${id}")" = 204
test "$(curl -s -o /dev/null -w '%{http_code}' "${api}/appointments/${id}")" = 404
curl -fsS "${api}/stats/today" | jq .
echo "Compose end-to-end OK"

#!/bin/bash
set -eo pipefail
echo "=== step: Image name (verbatim) ==="
image="ghcr.io/${GITHUB_REPOSITORY_OWNER,,}/carequeue-${COMPONENT}"
echo "IMAGE=${image}" >> "$GITHUB_ENV"
echo "IMAGE_REF=${image}:${GITHUB_SHA}" >> "$GITHUB_ENV"
echo "Image: ${image}:${GITHUB_SHA}"

echo "=== step: Inspect image (size, user) (verbatim) ==="
set -a; . "$GITHUB_ENV"; set +a
docker image ls "${IMAGE}"
user="$(docker image inspect -f '{{.Config.User}}' "${IMAGE_REF}")"
echo "Image user: ${user:-<root!>}"

echo "=== step: Trivy reports (all severities, JSON + SARIF, never fails) (verbatim) ==="
set -a; . "$GITHUB_ENV"; set +a
mkdir -p reports
trivy image --scanners vuln --format json  -o "reports/trivy-${COMPONENT}.json"  "${IMAGE_REF}"
trivy image --scanners vuln --format sarif -o "reports/trivy-${COMPONENT}.sarif" --skip-db-update "${IMAGE_REF}"

echo "=== step: Trivy scan summary (verbatim) ==="
json="reports/trivy-${COMPONENT}.json"
{
  echo "### Trivy image scan: carequeue-${COMPONENT}"
  echo ""
  echo "Image: \`${IMAGE_REF}\`"
  echo ""
  echo "OS: $(jq -r '.Metadata.OS | "\(.Family // "?") \(.Name // "")"' "${json}")"
  echo ""
  echo "| Severity | All | Fixable |"
  echo "|---|---|---|"
  for s in CRITICAL HIGH MEDIUM LOW UNKNOWN; do
    all=$(jq --arg s "$s" '[.Results[]?.Vulnerabilities[]? | select(.Severity == $s)] | length' "${json}")
    fix=$(jq --arg s "$s" '[.Results[]?.Vulnerabilities[]? | select(.Severity == $s and (.FixedVersion // "") != "")] | length' "${json}")
    echo "| ${s} | ${all} | ${fix} |"
  done
  echo ""
  echo "Gate: fail on fixable HIGH/CRITICAL (\`--severity HIGH,CRITICAL --ignore-unfixed --exit-code 1\`)."
} >> "$GITHUB_STEP_SUMMARY"

echo "=== step: Trivy gate (fixable HIGH/CRITICAL fail the pipeline) (verbatim) ==="
trivy image --scanners vuln --skip-db-update \
  --severity HIGH,CRITICAL --ignore-unfixed --exit-code 1 \
  --format table "${IMAGE_REF}"

$ # --- step: Install ingress-nginx (verbatim) ---
  | kubectl apply -f https://raw.githubusercontent.com/kubernetes/ingress-nginx/controller-v1.15.1/deploy/static/provider/kind/deploy.yaml
  | # rollout status (not "wait --for=condition=Available": with 1 replica that
  | # condition is already true before the pod is Ready).
  | kubectl -n ingress-nginx rollout status deployment/ingress-nginx-controller --timeout=180s
  | # The admission webhook can still refuse for a few seconds after the pod is
  | # Ready, and an Ingress applied in that gap is rejected: retry a server-side dry run.
  | probe='{"apiVersion":"networking.k8s.io/v1","kind":"Ingress","metadata":{"name":"webhook-probe","namespace":"default"},"spec":{"ingressClassName":"nginx","defaultBackend":{"service":{"name":"none","port":{"number":80}}}}}'
  | for i in $(seq 1 40); do
  |   if echo "${probe}" | kubectl apply --dry-run=server -f - >/dev/null 2>&1; then
  |     echo "ingress-nginx admission webhook ready after ~$((i * 3))s"; break
  |   fi
  |   if [ "${i}" = 40 ]; then echo "admission webhook never became ready"; exit 1; fi
  |   sleep 3
  | done
